USE faculty_db;

-- ============================================================
-- MEMBER 2 – Attendance & Medical
-- ============================================================

-- One row per subject per student per session
DROP TABLE IF EXISTS attendance;
CREATE TABLE attendance (
    id           INT AUTO_INCREMENT PRIMARY KEY,
    student_id   VARCHAR(20) NOT NULL,          -- e.g. TG1701
    course_code  VARCHAR(20) NOT NULL,           -- e.g. ICT2132
    session_no   INT         NOT NULL,           -- 1 – 15
    session_type VARCHAR(10) NOT NULL,           -- THEORY or PRACTICAL
    is_present   TINYINT(1)  NOT NULL DEFAULT 0, -- 1=present, 0=absent
    session_date DATE        NOT NULL
);

-- Medical records submitted by students / entered by TO
DROP TABLE IF EXISTS medicals;
CREATE TABLE medicals (
    id           INT AUTO_INCREMENT PRIMARY KEY,
    student_id   VARCHAR(20) NOT NULL,
    from_date    DATE        NOT NULL,
    to_date      DATE        NOT NULL,
    reason       TEXT,
    is_approved  TINYINT(1)  DEFAULT 0,          -- 0=pending, 1=approved
    submitted_at TIMESTAMP   DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- MEMBER 3 – Marks & Grading
-- ============================================================

-- Marks per student per course (CA and Final separately)
DROP TABLE IF EXISTS marks;
CREATE TABLE marks (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    student_id  VARCHAR(20) NOT NULL,
    course_code VARCHAR(20) NOT NULL,
    ca_marks    DOUBLE DEFAULT 0,    -- Continuous Assessment (out of 100)
    final_marks DOUBLE DEFAULT 0,   -- Final Exam (out of 100)
    UNIQUE KEY uq_student_course (student_id, course_code)
);

-- ============================================================
-- MEMBER 2 – Sample Attendance Data (covers all 5 scenarios)
-- ============================================================

-- Helper: sessions 1-15 for each course
-- THEORY sessions
INSERT INTO attendance (student_id, course_code, session_no, session_type, is_present, session_date)
SELECT
    u.username,
    c.course_code,
    s.n,
    'THEORY',
    CASE
        -- TG1701 & TG1702: >80% (attend 13/15 = 86%)
        WHEN u.username IN ('TG1701','TG1702') AND s.n <= 13 THEN 1
        WHEN u.username IN ('TG1701','TG1702') AND s.n >  13 THEN 0
        -- TG17013 & TG1704: exactly 80% (attend 12/15)
        WHEN u.username IN ('TG17013','TG1704') AND s.n <= 12 THEN 1
        WHEN u.username IN ('TG17013','TG1704') AND s.n >  12 THEN 0
        -- TG1705 & TG1706: <80% no medicals (attend 10/15 = 66%)
        WHEN u.username IN ('TG1705','TG1706') AND s.n <= 10 THEN 1
        WHEN u.username IN ('TG1705','TG1706') AND s.n >  10 THEN 0
        -- TG1707 & TG1708: <80% but WITH medicals (attend 10/15 raw)
        WHEN u.username IN ('TG1707','TG1708') AND s.n <= 10 THEN 1
        WHEN u.username IN ('TG1707','TG1708') AND s.n >  10 THEN 0
        -- TG1709–TG1720: >80% with medicals (attend 11/15, medical covers 2 absences)
        WHEN s.n <= 11 THEN 1
        ELSE 0
    END,
    DATE_ADD('2026-01-05', INTERVAL (s.n - 1) * 7 DAY)
FROM
    users u
    JOIN courses c ON 1=1
    JOIN (SELECT 1 n UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5
          UNION SELECT 6 UNION SELECT 7 UNION SELECT 8 UNION SELECT 9 UNION SELECT 10
          UNION SELECT 11 UNION SELECT 12 UNION SELECT 13 UNION SELECT 14 UNION SELECT 15) s
WHERE u.role = 'STUDENT';
